def calcular_split_payment_impacto(self) -> Dict:
    ano = self.operacao.data_emissao.year
    forma = self.operacao.forma_recebimento

    if ano >= ANO_INICIO_SPLIT_PAYMENT and forma != "DINHEIRO":     # ← if 1
        aliquotas_ano = self.get_aliquotas_iva_por_ano()            # ← busca dict
        taxa_retencao = aliquotas_ano["CBS"] + aliquotas_ano["IBS"]  # ← mescla tipos
        retencao = (self.operacao.valor_operacao * taxa_retencao)    # ← float risk
        return {
            "ativo": True,
            "retencao_imediata": str(retencao),
            ...
        }

    return {                                                          # ← if 2
        "ativo": False,
        "motivo": (                                                  # ← nested?
            f"Ano {ano} < {ANO_INICIO_SPLIT_PAYMENT} ..."
            if ano < ANO_INICIO_SPLIT_PAYMENT
            else "Pagamento em DINHEIRO: ..."
        ),
        ...
    }

PROBLEMAS:
  • 14 linhas com if/else aninhados
  • Mix de tipos (dict["CBS"] vs. Decimal)
  • Sem rastreabilidade de lei
  • Hard para estender (novo if = 1h)
  • 60% de cobertura de teste


DEPOIS (Strategy Pattern):
══════════════════════════

def calcular_split_payment_impacto(self) -> Dict:
    calculadora = CalculadoraSplitPayment()
    resultado = calculadora.calcular(
        ParametrosSplitPayment(
            valor_nf=self.operacao.valor_operacao,
            ano_emissao=self.operacao.data_emissao.year,
            forma_recebimento=FormaRecebimento[self.operacao.forma_recebimento]
        )
    )

    return {
        "ativo": resultado.ativo,
        "retencao_imediata": str(resultado.valor_retido),
        "percentual_retencao": f"{resultado.percentual_taxa * 100:.2f}%",
        "lei_vigente": resultado.lei_vigente,
        "estrategia": resultado.estrategia_classe,
        "impacto_liquidez": resultado.mensagem_impacto,
        "avisos": resultado.avisos or [],
    }

BENEFÍCIOS:
  • 3 linhas de lógica clara
  • Tipos estruturados (Dataclass + Enum)
  • Lei vinculada ao ano (Enum.lei_vigente)
  • Extensível sem editar lógica (nova Strategy)
  • 100% de cobertura de teste

FLUXO DE DADOS: PASSO A PASSO (2027, PIX, R$ 10.000)
================================================================================

1. ENTRADA
   valor_nf=10000.00, ano=2027, forma=PIX

2. VALIDAÇÃO (Enum)
   AnoTransicaoIVA.from_ano(2027)
   → ANO_2027 (lei_vigente="LC 214/2025, Art. 344-353", fase="EFETIVO")

3. BUSCA DE ALÍQUOTAS (Registry)
   CronogramaIVA.obter_aliquotas(2027)
   → AliquotasIVA(cbs=0.088, ibs=0.001, total=0.089)

4. SELEÇÃO DE ESTRATÉGIA (Factory)
   FabricaEstrategiaRetencao.obter_estrategia(2027, PIX, 0.0)
   Lógica:
     • ano==2026? NÃO
     • not forma.eh_eletronico()? NÃO (PIX é eletrônico)
     • politica_reducao>0? NÃO
     → RetencaoPadrao

5. EXECUÇÃO (Strategy)
   RetencaoPadrao.calcular_retencao(10000, AliquotasIVA, PIX)
   = 10000 × (0.088 + 0.001)
   = 10000 × 0.089
   = 890.00 (quantized Decimal)

6. ELEGIBILIDADE (Strategy)
   RetencaoPadrao.eh_elegivel(2027, PIX)
   = (2027 >= 2027) AND (PIX.eh_eletronico())
   = True

7. MENSAGEM (Strategy)
   RetencaoPadrao.obter_mensagem_impacto()
   = "Split Payment padrão (LC 214/2025): retenção integral na fonte. ..."

8. MONTAGEM DE RESULTADO (Dataclass)
   ResultadoSplitPayment(
       ativo=True,
       valor_retido=Decimal("890.00"),
       percentual_taxa=Decimal("0.089"),

class RetencaoSplitPaymentPME:
    def __init__(self, fator_reducao: Decimal = Decimal("0.50")):
        self.fator_reducao = fator_reducao

    def calcular_retencao(self, valor_nf, aliq, forma):
        taxa_reduzida = aliq.total() * (Decimal("1.0") - self.fator_reducao)
        return (valor_nf * taxa_reduzida).quantize(Decimal("0.01"), ROUND_HALF_UP)

    def eh_elegivel(self, ano, forma):
        return ano >= 2027 and forma.eh_eletronico()

    def obter_mensagem_impacto(self):
        return f"Split Payment PME (redução {self.fator_reducao*100:.0f}%)..."
